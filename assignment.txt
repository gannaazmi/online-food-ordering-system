create database prodductDB;

create table product(
maker varchar(20),
model varchar(20) primary key, 
type varchar(20)
);

create table PC(
model varchar(20) primary key,
speed int,
RAM int,
HD int,
RD varchar(20),
price decimal(10,4),
foreign key (model) references product(model)
);

create table laptop(
model varchar(20) primary key,
speed int,
RAM int,
HD int,
screen decimal (4,1),
price decimal(10,4),
FOREIGN KEY (model) REFERENCES product(model)
);

create table printer(
model varchar(20) primary key,
color varchar(20),
type varchar(20),
price decimal(10,4),
FOREIGN KEY (model) REFERENCES product(model)
);

select p.model,pr.price
from product p
join printer pr on p.model=pr.model
where p.maker='B'
